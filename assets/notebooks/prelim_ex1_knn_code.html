from sklearn.model_selection import train_test_split
X_Train, X_Test, y_Train, y_Test = train_test_split(X, y, test_size=0.2, random_state=0)

# Visualization of the Scatter plot with predicted classes
import matplotlib.pyplot as plt
plt.figure(figsize=(8,6))

# Define colors for true classes (mimicking viridis for first three classes visually)
colors = ['#440154', '#21908d', '#fde725'] # Example colors from viridis cmap for distinctness

for i in range(3):
  plt.scatter(X_Test[:,0][y_Test == i], X_Test[:,1][y_Test ==i], c=colors[i], label=data.target_names[i])

# Plot the predicted classes with original viridis colormap, removing marker and size
plt.scatter(X_Test[:,0],X_Test[:,1], c=predictions, cmap='viridis', alpha=0.8, label='Predicted Classes')

plt.xlabel(data.feature_names[0])
plt.ylabel(data.feature_names[1])
plt.title("Scatter plot with true and predicted classes")
plt.legend()
plt.show()
print()

# Making a confusion matrix
from sklearn.metrics import confusion_matrix
cm = confusion_matrix(y_Test, predictions)

# Confusion matrix visualization
plt.figure(figsize=(8, 6))
plt.imshow(cm, cmap='Blues' )
plt.colorbar()
plt.title("Confusion Matrix")
plt.xlabel("Predicted Label")
plt.ylabel("True Label")
plt.xticks(ticks=range(3), labels=data.target_names)
plt.yticks(ticks=range(3), labels=data.target_names)
plt.show()
